local function injectStarfield(gui)
    if not gui then return end
    local biggest, biggestArea = nil, 0
    for _, d in pairs(gui:GetDescendants()) do
        if d:IsA("Frame") and d.BackgroundTransparency < 0.6 then
            local a = d.AbsoluteSize.X * d.AbsoluteSize.Y
            if a > biggestArea then
                biggestArea = a
                biggest = d
            end
        end
    end
    if biggest and not biggest:GetAttribute("PolyStarfieldDone") then
        createStarField(biggest)
    end
end

local function isAccentPurpleBlue(c)
    local h, s, v = Color3.toHSV(c)
    return h >= 0.55 and h <= 0.88 and s > 0.35 and v > 0.4
end

local function isRed(c)
    local h, s, v = Color3.toHSV(c)
    return (h < 0.05 or h > 0.95) and s > 0.4 and v > 0.4
end

local function isYellow(c)
    local h, s, v = Color3.toHSV(c)
    return h > 0.08 and h < 0.20 and s > 0.4 and v > 0.5
end

local function recolor()
    local gui = findGui(); if not gui then return end
    for _, o in ipairs(gui:GetDescendants()) do
        pcall(function()
            local n = o.Name
            if n == "PolythiumTitleLogo" or n == "PolyStarfield" then return end

            if o:IsA("Frame") or o:IsA("TextButton") or o:IsA("ImageButton") then
                local bg = o.BackgroundColor3
                local small = o.Size.X.Offset <= 26 and o.Size.Y.Offset <= 26

                if isAccentPurpleBlue(bg) or isRed(bg) then
                    o.BackgroundColor3 = small and PURPLE or BLUE
                elseif isYellow(bg) then
                    o.BackgroundColor3 = PURPLE
                end

                if o.Parent and o.Parent:IsA("Frame") then
                    local pbg = o.Parent.BackgroundColor3
                    local parentRed = (pbg.R > 0.6 and pbg.G < 0.5 and pbg.B < 0.5)
                    local parentBlue = (pbg.B > 0.6 and pbg.R < 0.6)
                    local isWhite = bg.R > 0.9 and bg.G > 0.9 and bg.B > 0.9
                    if (parentRed or parentBlue) and isWhite and small then
                        o.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
                    end
                end
            end

            if o:IsA("ImageLabel") and (isAccentPurpleBlue(o.ImageColor3) or isRed(o.ImageColor3)) then
                o.ImageColor3 = PURPLE
            end
            if o:IsA("UIStroke") and (isAccentPurpleBlue(o.Color) or isRed(o.Color)) then
                o.Color = PURPLE
            end
            if (o:IsA("TextLabel") or o:IsA("TextButton")) then
                if isAccentPurpleBlue(o.TextColor3) or isRed(o.TextColor3) then
                    o.TextColor3 = PURPLE
                elseif isYellow(o.TextColor3) then
                    o.TextColor3 = PURPLE_LT
                end
            end
        end)
    end
    injectStarfield(gui)
end

end

-- ---------- Самокалибровка предсказания ----------
-- Каждый кадр сравниваем реальное ускорение мяча с модельным (гравитация + сопротивление).
-- Разницу (кручение, нестандартная гравитация, силы игры) запоминаем и добавляем в симуляцию,
-- поэтому прогноз быстро подстраивается под реальную физику мяча.
local function knownAccel(v)
    return Vector3.new(0, -workspace.Gravity, 0) - v * (CFG.AirDrag * v.Magnitude)
end

local function updateTrack(ball)
    local now = os.clock()
    local v = ball.AssemblyLinearVelocity
    local tr = track[ball]
    if not tr then track[ball] = {v = v, t = now, res = Vector3.zero} return end
    if (v - tr.v).Magnitude < 1e-3 then return end   -- физика ещё не обновилась

    local lv, dt = tr.v, now - tr.t
    tr.v, tr.t = v, now
    if dt < 0.004 or dt > 0.25 or v.Magnitude < 1 then tr.res = Vector3.zero return end

    local radius = math.max(ball.Size.X, ball.Size.Y, ball.Size.Z) / 2
    if workspace:Raycast(ball.Position, Vector3.new(0, -(radius + 0.6), 0), rayParams) then
        tr.res = Vector3.zero   -- мяч на земле: нормальная сила, не калибруем
        return
    end

    local c = (v - lv) / dt - knownAccel(lv)
    if c.Magnitude > 120 then tr.res = Vector3.zero return end   -- удар/столкновение
    tr.res = tr.res:Lerp(c, CFG.Adapt)
end

local function simulate(ball, predictTime)
    refreshRayParams()
    local pos, vel = ball.Position, ball.AssemblyLinearVelocity
    local points, times = { pos }, { 0 }
    if vel.Magnitude < 0.5 then return points, times, pos end

    local tr = track[ball]
    local acc = Vector3.new(0, -workspace.Gravity, 0) + (tr and tr.res or Vector3.zero)
    local radius = math.max(ball.Size.X, ball.Size.Y, ball.Size.Z) / 2
    local dt = CFG.SimDt
    local landing
    local bounces, t = 0, 0

    for _ = 1, math.floor((predictTime or CFG.PredictTime) / dt) do
        local v = vel + acc * dt
        v = v * math.max(0, 1 - CFG.AirDrag * v.Magnitude * dt)
        local step = v * dt
        local nextPos = pos + step
        local hit
        if step.Magnitude > 0 then
            hit = workspace:Raycast(pos, step + step.Unit * radius, rayParams)
        end
        t += dt

        if hit then
            local n = hit.Normal.Unit
            local vn = v:Dot(n)
            if vn < 0 then
                local vN = n * vn
                local vT = v - vN
                if -vn < 4 then
                    v = vT * math.max(0, 1 - CFG.RollFriction * dt)   -- катится
                else
                    v = vT * CFG.GroundFriction - vN * CFG.Restitution -- отскок
                    bounces += 1
                    if n.Y > 0.7 and not landing then landing = hit.Position end
                end
            end
            nextPos = hit.Position + n * radius
            table.insert(points, nextPos) table.insert(times, t)
            if v.Magnitude < 2.5 or bounces > CFG.MaxBounces then break end
        else
            table.insert(points, nextPos) table.insert(times, t)
        end
        pos, vel = nextPos, v
    end
    return points, times, landing or points[#points]
end

-- ---------- Отрисовка траектории ----------
local function createSet()
    local markers = {}
    for i = 1, CFG.NumMarkers do
        local m = newPart(Enum.PartType.Ball, Vector3.one * CFG.MarkerSize)
        m.Color = CFG.TrajColor
        markers[i] = m
    end
    local landing = newPart(Enum.PartType.Cylinder, Vector3.new(0.08, CFG.LandingSize, CFG.LandingSize))
    landing.Color = CFG.LandingColor
    return { markers = markers, landing = landing }
end

local function destroySet(set)
    for _, m in ipairs(set.markers) do if m.Parent then m:Destroy() end end
    if set.landing and set.landing.Parent then set.landing:Destroy() end
end

local targetMarker
local function clearAll()
    for _, set in pairs(trajBalls) do destroySet(set) end
    trajBalls = {}
    targetMarker = nil
    if trajFolder then trajFolder:Destroy() trajFolder = nil end
end

local function rescan()
    local active = {}
    for _, b in ipairs(getBalls()) do active[b] = true end
    for b in pairs(active) do if not trajBalls[b] then trajBalls[b] = createSet() end end
    for b, set in pairs(trajBalls) do
        if not b.Parent or not active[b] then destroySet(set) trajBalls[b] = nil end
    end
    for b in pairs(track) do if not b.Parent then track[b] = nil end end
end

local function drawTraj(ball, set)
    local pts, _, landing = simulate(ball)
    if #pts < 2 then
        for _, m in ipairs(set.markers) do m.Transparency = 1 end
        set.landing.Transparency = 1
        return
    end
    local step = math.max(1, math.floor(#pts / CFG.NumMarkers))
    for i = 1, CFG.NumMarkers do
        local idx = (i - 1) * step + 1
        local m = set.markers[i]
        if pts[idx] then
            local frac = i / CFG.NumMarkers
            m.Position = pts[idx]
            m.Transparency = 0.05 + 0.55 * frac
            m.Color = CFG.TrajColor
            m.Size = Vector3.one * CFG.MarkerSize * (1 - 0.4 * frac)
        else
            m.Transparency = 1
        end
    end
    set.landing.CFrame = CFrame.new(landing + Vector3.new(0, 0.04, 0)) * CFrame.Angles(0, 0, math.rad(90))
    set.landing.Color = CFG.LandingColor
    set.landing.Transparency = 0.15
end

local lastCatch, lastTouch = 0, 0
local catchWindow = { untilT = 0, ball = nil }

local function reachTouch(char, tool, ball)
    if typeof(firetouchinterest) ~= "function" then return end
    local handle = tool:FindFirstChild("Handle")
    if handle and handle:IsA("BasePart") then
        pcall(firetouchinterest, handle, ball, 0)
        pcall(firetouchinterest, handle, ball, 1)
    end
    for _, part in ipairs(char:GetChildren()) do
        if part:IsA("BasePart") then
            pcall(firetouchinterest, part, ball, 0)
            pcall(firetouchinterest, part, ball, 1)
        end
    end
end

local function autoCatch()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local gk = getEquippedGK(char)
    if not (hrp and gk) then catchWindow.untilT = 0 return end   -- без GK ничего не делаем

    local now = tick()

    -- окно ловли: пока оно открыто, "касаемся" мяча в пределах reach
    if CFG.ReachTouch and now < catchWindow.untilT and catchWindow.ball and catchWindow.ball.Parent
        and now - lastTouch > 0.03 then
        local ball = catchWindow.ball
        if (ball.Position - hrp.Position).Magnitude <= CFG.CatchReach * 1.25 then
            lastTouch = now
            reachTouch(char, gk, ball)
        end
    end

    if now - lastCatch < CFG.AbilityCooldown then return end

    local bestBall, bestT, bestY
    for _, ball in ipairs(getBalls()) do
        local vel = ball.AssemblyLinearVelocity
        if vel.Magnitude >= CFG.CatchMinSpeed then
            local toMe = hrp.Position - ball.Position
            if toMe.Magnitude < 6 or vel.Unit:Dot(toMe.Unit) > 0.1 then
                -- точка максимального сближения с персонажем в ближайшие CatchLookahead сек
                local pts, times = simulate(ball, CFG.CatchLookahead + 0.05)
                local minD, minI = math.huge, 1
                for i, p in ipairs(pts) do
                    if times[i] > CFG.CatchLookahead then break end
                    local d = (p - hrp.Position).Magnitude
                    if d < minD then minD, minI = d, i end
                end
                if minD <= CFG.CatchReach and (not bestT or times[minI] < bestT) then
                    bestBall, bestT, bestY = ball, times[minI], pts[minI].Y
                end
            end
        end
    end

    if bestBall then
        lastCatch = now
        local high = (bestY - hrp.Position.Y) > CFG.HighThreshold
        pressKey(high and "R" or "F")
        catchWindow.untilT = now + CFG.CatchWindow
        catchWindow.ball = bestBall
        if CFG.ReachTouch then reachTouch(char, gk, bestBall) end
    end
end

-- ---------- AutoSave (только чужие удары) ----------
local shots, hist, decided = {}, {}, {}
local saveTarget, saveBall, saveMoving = nil, nil, false

local function hideTarget() if targetMarker then targetMarker.Transparency = 1 end end

local function nearestShooterIsMe(ball)
    local myHrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not myHrp then return false end
    local myD = (myHrp.Position - ball.Position).Magnitude
    local otherD = math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            local h = plr.Character:FindFirstChild("HumanoidRootPart")
            if h then otherD = math.min(otherD, (h.Position - ball.Position).Magnitude) end
        end
    end
    return myD <= otherD
end

local function detectShots()
    local now = tick()
    for _, ball in ipairs(getBalls()) do
        local sp = ball.AssemblyLinearVelocity.Magnitude
        local h = hist[ball]
        if not h then h = {} hist[ball] = h end
        table.insert(h, {now, sp})
        while #h > 0 and now - h[1][1] > 0.12 do table.remove(h, 1) end

        if sp >= CFG.ShotMinSpeed and #h >= 2 then
            local minPrev = math.huge
            for i = 1, #h - 1 do minPrev = math.min(minPrev, h[i][2]) end
            local d = decided[ball]
            if sp - minPrev >= CFG.ShotMinJump and (not d or now - d.t > 0.25) then
                local mine = nearestShooterIsMe(ball)
                decided[ball] = { t = now, mine = mine }
                shots[ball] = (not mine) and { t0 = now } or nil
            end
        end

        local sh = shots[ball]
        if sh and (now - sh.t0 > CFG.ShotMaxTime or sp < CFG.ShotMinSpeed * 0.3) then shots[ball] = nil end
    end
    for b in pairs(hist) do
        if not b.Parent then hist[b], shots[b], decided[b] = nil, nil, nil end
    end
end

-- Самая ранняя точка траектории, куда успеваем добежать и достать по высоте.
-- Если не успеваем никуда, бежим в точку, где отставание минимально.
local function findIntercept(ball, hrp, hum)
    local pts, times, landing = simulate(ball, CFG.SavePredict)
    local speed = math.max(hum.WalkSpeed, 1)
    local footY = hrp.Position.Y - 3
    local fallback, fallbackLag = nil, math.huge
    for i, p in ipairs(pts) do
        if p.Y - footY <= CFG.SaveMaxHeight then
            local flat = Vector3.new(p.X - hrp.Position.X, 0, p.Z - hrp.Position.Z).Magnitude
            local lag = flat / speed + CFG.ReactionTime - times[i]
            if lag <= 0 then return p end
            if lag < fallbackLag then fallback, fallbackLag = p, lag end
        end
    end
    return fallback or landing
end

local function stopRunning(hum, hrp)
    if saveMoving then
        saveMoving = false
        if hum and hrp then pcall(function() hum:MoveTo(hrp.Position) end) end
    end
    saveTarget, saveBall = nil, nil
end

local function autoSave()
    local char = LP.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (hrp and hum) then return end

    detectShots()

    local best, bestT
    for ball, sh in pairs(shots) do
        if ball.Parent then
            local toMe = hrp.Position - ball.Position
            local vel = ball.AssemblyLinearVelocity
            if toMe.Magnitude < 25 or (vel.Magnitude > 0 and vel.Unit:Dot(toMe.Unit) > 0.2) then
                if not bestT or sh.t0 > bestT then best, bestT = ball, sh.t0 end
            end
        end
    end
    if not best then hideTarget() stopRunning(hum, hrp) return end

    local target = findIntercept(best, hrp, hum)
    if not target or (target - hrp.Position).Magnitude > CFG.SaveRadius then
        hideTarget() stopRunning(hum, hrp) return
    end

    -- лёгкое сглаживание, чтобы цель не дёргалась между кадрами
    if saveBall ~= best or not saveTarget then saveTarget, saveBall = target, best
    else saveTarget = saveTarget:Lerp(target, 0.5) end

    if CFG.ShowTarget then
        if not targetMarker or not targetMarker.Parent then
            targetMarker = newPart(Enum.PartType.Ball, Vector3.one * 2)
            targetMarker.Color = Color3.fromRGB(255, 90, 90)
        end
        targetMarker.Position = saveTarget
        targetMarker.Transparency = 0.2
    else
        hideTarget()
    end

    saveMoving = true
    hum:MoveTo(Vector3.new(saveTarget.X, hrp.Position.Y, saveTarget.Z))
end
